# =============================================================================
# GLP tachanka branch anchor
# =============================================================================
#
# A technology with only `folder = { ... }` and no incoming `path` is not
# assigned a gridbox by the vanilla countrytechtreeview GUI.  Consequently
# GLP_tachanka_tech_1 (the root of the custom branch) was silently omitted
# from the infantry folder even though its technology definition was valid.
#
# HOI4 has no partial merge for an existing technology ID.  The block below
# therefore restates the vanilla "Infantry Weapons I" technology and adds an
# incoming path to our custom root.  The zzz_ prefix makes this definition
# load after the vanilla infantry technology file, so the restated block
# replaces the vanilla one (all vanilla fields are reproduced below).
#
# The tachanka line is anchored to infantry_weapons1 (the 1936 infantry
# weapons tech, the second icon in the infantry row).  The incoming path
# assigns the branch to the existing `infantry_weapons_tree` gridbox, so no
# GUI override is needed.  The GLP branch occupies the empty band between the
# infantry weapons row (y = 325) and the motorized/trucks row (y = 620):
# it starts directly below Infantry Weapons I (x = 2, y = 2) and continues
# right along the same band (x = 2, y = 2..5).  It is therefore clear of the
# special-forces rows (marines / mountaineers / paratroopers / rangers),
# which live in their own gridboxes at y = 825..1125.
#
# The custom root has allow_branch = { tag = GLP }, so the anchor path and
# the tachanka line are visible only in the GLP research tree.
#
# The vanilla infantry_weapons1 values below are the 1.19-compatible
# definition; do not remove the original enable/reward/AI fields when
# updating this anchor for a future game version.

technologies = {

	infantry_weapons1 = {
		enable_equipments = {
			infantry_equipment_1
		}

		path = {
			leads_to_tech = infantry_weapons2
			research_cost_coeff = 1
		}

		# Gridbox anchor for the GLP tachanka branch.
		path = {
			leads_to_tech = GLP_tachanka_tech_1
			research_cost_coeff = 1
		}

		research_cost = 1.5
		start_year = 1936
		folder = {
			name = infantry_folder
			position = { x = 0 y = 2 }
		}

		categories = {
			infantry_weapons
		}

		ai_will_do = {
			base = 2

			modifier = {
				factor = 2
				is_major = yes
			}

			modifier = {
				factor = 10
				date > 1936.1.3
			}
		}
	}
}
