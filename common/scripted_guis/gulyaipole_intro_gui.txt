# Гуляй-Поле — scripted GUI кинематографической заставки (HOI4 1.19.2)
#
# Окно из interface/gulyaipole_intro_custom.gui показывается, пока установлен
# glp_show_cinematic_intro. В центральной колонке — история Вольной республики.
# Кнопка «Вольному воля!» закрывает окно и плавно переключает музыку на следующий
# случайный трек из плейлиста «Монгол Шуудан» (чередуя композиции).

scripted_gui = {
	gulyaipole_cinematic_intro = {
		window_name = "gulyaipole_cinematic_intro_window"
		context_type = player_context

		visible = {
			tag = GLP
			is_ai = no
			has_country_flag = glp_show_cinematic_intro
		}

		effects = {
			start_campaign_button_click = {
				clr_country_flag = glp_show_cinematic_intro
				clr_country_flag = glp_intro_crawl_mode
				clr_country_flag = glp_intro_manual_mode
				random_list = {
					17 = { scoped_play_song = GLP_ms_lyubo }
					17 = { scoped_play_song = GLP_ms_bei }
					17 = { scoped_play_song = GLP_ms_sedmoi_patron }
					17 = { scoped_play_song = GLP_ms_konflikt }
					16 = { scoped_play_song = GLP_ms_finsky_nozh }
					16 = { scoped_play_song = GLP_ms_ne_moya_zabota }
				}
			}
			close_intro_button_click = {
				clr_country_flag = glp_show_cinematic_intro
				clr_country_flag = glp_intro_crawl_mode
				clr_country_flag = glp_intro_manual_mode
				random_list = {
					17 = { scoped_play_song = GLP_ms_lyubo }
					17 = { scoped_play_song = GLP_ms_bei }
					17 = { scoped_play_song = GLP_ms_sedmoi_patron }
					17 = { scoped_play_song = GLP_ms_konflikt }
					16 = { scoped_play_song = GLP_ms_finsky_nozh }
					16 = { scoped_play_song = GLP_ms_ne_moya_zabota }
				}
			}
		}

		triggers = {
			start_campaign_button_click_enabled = {
				always = yes
			}
			start_campaign_button_visible = {
				always = yes
			}
			close_intro_button_click_enabled = {
				always = yes
			}
			close_intro_button_visible = {
				always = yes
			}
		}
	}
}
