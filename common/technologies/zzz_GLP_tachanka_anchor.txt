# =============================================================================
# GLP tachanka branch anchor
# =============================================================================
#
# A technology with only `folder = { ... }` and no incoming `path` is not
# assigned a gridbox by the vanilla countrytechtreeview GUI.  Consequently
# GLP_tachanka_tech_1 (the root of the custom branch) was silently omitted
# from the infantry folder even though its technology definition was valid.
#
# HOI4 has no partial merge for an existing technology ID.  The block below
# therefore restates the vanilla paratroopers technology and adds an incoming
# path to our custom root.  The zzz_ prefix makes this definition load after
# the vanilla infantry technology file.  This reuses the existing
# `paratroopers_tree` gridbox and does not override the 6000+ line GUI file.
# The custom root has allow_branch = { tag = GLP }, so the anchor and the
# tachanka line are visible only in the GLP research tree.
#
# The vanilla paratroopers values below are the 1.19-compatible definition;
# do not remove the original enable/reward/AI fields when updating this
# anchor for a future game version.

technologies = {

	paratroopers = {
		enable_subunits = {
			paratrooper
		}
		enable_subunits = {
			limit = { has_dlc = "Thunder at Our Gates" }
			hq_paratrooper
		}

		path = {
			leads_to_tech = paratroopers2
			research_cost_coeff = 1
		}

		# Gridbox anchor for the GLP tachanka branch.
		path = {
			leads_to_tech = GLP_tachanka_tech_1
			research_cost_coeff = 1
		}

		research_cost = 2
		start_year = 1936
		folder = {
			name = infantry_folder
			position = { x = 3 y = 2 }
		}

		on_research_complete_limit = {
			NOT = {
				has_template_containing_unit = paratrooper
			}
		}
		on_research_complete = {
			hidden_effect = {
				load_oob = "unlock_paratroopers"
			}
			custom_effect_tooltip = UNLOCK_DIVISION_TEMPLATE

			if = {
				limit = { has_dlc = "Gotterdammerung" }
				custom_effect_tooltip = {
					localization_key = SP_UNLOCK_RAID
					RAID_TYPE = raid_type_mine_sabotage_raid
				}
				custom_effect_tooltip = {
					localization_key = SP_UNLOCK_RAID
					RAID_TYPE = raid_type_pathfinder_raid
				}
			}
		}

		categories = {
			para_tech
			cat_special_forces_generic
		}

		ai_will_do = {
			factor = 0.5

			modifier = {
				factor = 20
				tag = USA
				date > 1943.1.1
			}

			modifier = {
				factor = 2
				tag = GER
			}

			modifier = {
				factor = 5
				tag = ENG
			}

			modifier = {
				factor = 20
				tag = GER
				has_completed_focus = GER_anschluss
				CZE = {
					is_ai = no
				}
			}

			modifier = {
				factor = 5
				tag = GER
				has_completed_focus = GER_fallschirmjager
			}
		}
	}
}
