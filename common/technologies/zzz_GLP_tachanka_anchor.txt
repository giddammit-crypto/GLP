# =============================================================================
# GLP tachanka branch anchor
# =============================================================================
#
# A technology with only `folder = { ... }` and no incoming `path` is not
# assigned a gridbox by the vanilla countrytechtreeview GUI.  Consequently
# GLP_tachanka_tech_1 (the root of the custom branch) was silently omitted
# from the infantry folder even though its technology definition was valid.
#
# HOI4 has no partial merge for an existing technology ID.  The block below
# therefore restates the vanilla "Motorized Trucks" technology and adds an
# incoming path to our custom root.  The zzz_ prefix makes this definition
# load after the vanilla infantry technology file, so the restated block
# replaces the vanilla one (all vanilla fields are reproduced below).
#
# The tachanka line is anchored to tech_trucks so that it inherits the
# existing `tech_trucks_tree` gridbox (the motorized row of the infantry
# folder, origin x = 140 y = 620) -- no GUI override is needed.  The anchor
# deliberately uses tech_trucks rather than armored_car1 because the vanilla
# armored cars are gated behind the "La Resistance" DLC via allow_branch and
# the tachanka branch must be visible for GLP without any DLC.
#
# Layout: the vanilla armored cars sit in the row y = 900 (armored_car1/2/3
# at pixel x = 210/700/980) and the marines row starts at y = 965.  The GLP
# tachanka branch is placed on row x = 6 (pixel row y = 1040), far to the
# right at y = 15/17/19/21 (pixel columns x = 1190/1330/1470/1610), using
# the same 140 px (2-slot) spacing as vanilla infantry research items.  This
# keeps the branch clear of the motorized row (y = 620), the armored cars
# (y = 900), the marine row (y = 965) and the other special forces rows
# below, while the four nodes no longer overlap each other.
#
# The custom root has allow_branch = { tag = GLP }, so the anchor path and
# the tachanka line are visible only in the GLP research tree.
#
# The vanilla tech_trucks values below are the 1.19-compatible definition;
# do not remove the original enable/reward/AI fields when updating this
# anchor for a future game version.

technologies = {

	tech_trucks = {
		enable_subunits = {
			mot_anti_tank_brigade
			mot_artillery_brigade
			mot_anti_air_brigade
		}
		enable_subunits = {
			limit = { has_dlc = "Thunder at Our Gates" }
			hq_motorized
		}

		enable_equipments = {
			motorized_equipment_0
		}

		enable_equipment_modules = {
			tank_wheeled_suspension
		}

		path = {
			leads_to_tech = motorised_infantry
			research_cost_coeff = 1
		}

		path = {
			leads_to_tech = armored_car1
			research_cost_coeff = 1
		}

		# Gridbox anchor for the GLP tachanka branch.
		path = {
			leads_to_tech = GLP_tachanka_tech_1
			research_cost_coeff = 1
		}

		research_cost = 1.5
		start_year = 1936
		folder = {
			name = infantry_folder
			position = { x = 0 y = -1 }
		}

		categories = {
			motorized_equipment
			mio_cat_tech_all_motorized_mechanized
			infantry_tech
		}
		special_project_specialization = { specialization_land }

		ai_will_do = {
			factor = 1

			modifier = {
				factor = 2
				date > "1937.1.1"
			}

			modifier = {
				factor = 2
				date > "1938.1.1"
			}

			modifier = {
				factor = 2
				date > "1939.1.1"
			}
			modifier = {
				is_major = no
				factor = 0.75
			}
		}
	}
}
