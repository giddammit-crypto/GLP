on_actions = {
	on_startup = {
		effect = {
			# --------------------------------------------------------------
			#  Наведение порядка на Вольной территории при старте партии.
			#
			#  1) Все стейты, принадлежащие ГЛП, немедленно переходят под
			#     её же контроль — никаких «чужих» флагов на карте.
			#  2) Дивизии любых других держав (в первую очередь советские
			#     соединения из ванильного OOB СССР, чьи провинции переданы
			#     Гуляйполю) телепортируются в собственные стейты владельца:
			#     на территории РПА не остаётся иностранных войск.
			#  Все дивизии самой РПА, наоборот, размещены строго в границах
			#  Вольной территории (см. history/units/GLP_1936.txt) — армии
			#  «в изгнании» в моде нет.
			# --------------------------------------------------------------
			GLP = {
				every_owned_state = {
					transfer_state_to = GLP
					teleport_armies = {
						limit = {
							NOT = { tag = GLP }
						}
						to_state_array = owned_controlled_states
					}
				}
				# ----------------------------------------------------------
				#  Убрать ванильных generic-советников. Ванильный
				#  history/general/generic_advisors.txt генерирует для всех
				#  стран вне списка исключений случайных советников с именами
				#  из common/names (отсюда «Дмитрий Щусь», «Михаил Шевченко»)
				#  и ванильными портретами. Токен сгенерированного персонажа
				#  детерминирован: <TAG>_<token_base>. Каждый retire обёрнут
				#  в has_character (документированный триггер 1.11+), поэтому
				#  отсутствие персонажа (нет DLC/другая версия) не даёт ошибок.
				#  У Гуляйполя остаются только собственные 16 персонажей
				#  (плюс союзные военные миссии и белые эмигранты из фокусов).
				# ----------------------------------------------------------
				if = { limit = { has_character = GLP_generic_communist_revolutionary }	retire_character = GLP_generic_communist_revolutionary }
				if = { limit = { has_character = GLP_generic_democratic_reformer }		retire_character = GLP_generic_democratic_reformer }
				if = { limit = { has_character = GLP_generic_fascist_demagogue }			retire_character = GLP_generic_fascist_demagogue }
				if = { limit = { has_character = GLP_generic_head_of_intelligence }		retire_character = GLP_generic_head_of_intelligence }
				if = { limit = { has_character = GLP_generic_captain_of_industry }		retire_character = GLP_generic_captain_of_industry }
				if = { limit = { has_character = GLP_generic_fortification_engineer }	retire_character = GLP_generic_fortification_engineer }
				if = { limit = { has_character = GLP_generic_war_industrialist }			retire_character = GLP_generic_war_industrialist }
				if = { limit = { has_character = GLP_generic_army_chief_off }			retire_character = GLP_generic_army_chief_off }
				if = { limit = { has_character = GLP_generic_army_chief_def }			retire_character = GLP_generic_army_chief_def }
				if = { limit = { has_character = GLP_generic_navy_chief_decisive_bat }	retire_character = GLP_generic_navy_chief_decisive_bat }
				if = { limit = { has_character = GLP_generic_navy_anti_submarine }		retire_character = GLP_generic_navy_anti_submarine }
				if = { limit = { has_character = GLP_generic_air_close_air_sup }			retire_character = GLP_generic_air_close_air_sup }
				if = { limit = { has_character = GLP_generic_air_chief_all_weather }		retire_character = GLP_generic_air_chief_all_weather }
				if = { limit = { has_character = GLP_generic_army_art }					retire_character = GLP_generic_army_art }
				if = { limit = { has_character = GLP_generic_army_log }					retire_character = GLP_generic_army_log }
				if = { limit = { has_character = GLP_generic_navy_fleet_log }			retire_character = GLP_generic_navy_fleet_log }
				if = { limit = { has_character = GLP_generic_air_air_combat_trainer }	retire_character = GLP_generic_air_air_combat_trainer }
				if = { limit = { has_character = GLP_generic_military_theorist }			retire_character = GLP_generic_military_theorist }
				if = { limit = { has_character = GLP_generic_naval_theorist }			retire_character = GLP_generic_naval_theorist }
				if = { limit = { has_character = GLP_generic_air_warfare_theorist }		retire_character = GLP_generic_air_warfare_theorist }
				# ----------------------------------------------------------
				#  «Трофейный отряд РПА» — шаблон с РЕМОНТНОЙ РОТОЙ, из
				#  которого позже формируются новые дивизии из трофеев.
				#  Создаётся один раз (гард has_template при каждой
				#  загрузке сохранения не дублирует шаблон).
				# ----------------------------------------------------------
				if = {
					limit = { NOT = { has_template = "Трофейный отряд РПА" } }
					division_template = {
						name = "Трофейный отряд РПА"
						division_names_group = GLP_CAV_HISTORICAL
						is_locked = yes
						priority = 0
						regiments = {
							infantry = { x = 0 y = 0 }
							infantry = { x = 0 y = 1 }
							cavalry = { x = 0 y = 2 }
							cavalry = { x = 1 y = 0 }
						}
						support = {
							maintenance_company = { x = 0 y = 0 }
						}
					}
				}
				# ------------------------------------------------------------------
				#  Кинематографическая заставка Гуляй-Поля (HOI4 v1.19.2).
				#  Событие ставит флаг и запускает voice.ogg через sound_effect.
				#  Scripted GUI после этого показывает одно центрированное окно.
				# ------------------------------------------------------------------
				country_event = { id = glp_cinematic_intro.1 }

				# вступительная газета новой партіи — сразу послѣ загрузки
				news_event = { id = glp_news.100 hours = 2 }
				country_event = { id = glp_news.1 days = 2 }
				country_event = { id = glp_crisis.1 days = 150 }
				country_event = { id = glp_crisis.10 days = 220 }
				country_event = { id = glp_crisis.30 days = 900 }
				# Соседи сразу видят Вольную территорию как проблему, не как обычный минор.
				SOV = {
					add_ai_strategy = { type = antagonize id = "GLP" value = 160 }
					add_ai_strategy = { type = contain id = "GLP" value = 80 }
					add_ai_strategy = { type = conquer id = "GLP" value = 40 }
				}
				POL = {
					add_ai_strategy = { type = contain id = "GLP" value = 40 }
					add_ai_strategy = { type = antagonize id = "GLP" value = 25 }
				}
				ROM = {
					add_ai_strategy = { type = contain id = "GLP" value = 35 }
				}
				TUR = {
					add_ai_strategy = { type = contain id = "GLP" value = 20 }
				}
			}
		}
	}

	# ==============================================================
	#  ТРОФЕЙНЫЕ КОМАНДЫ РПА.
	#  При каждой победе войск Махно (любого командира GLP) в бою:
	#    * 65%  — трофейное стрелковое вооружение и снаряжение;
	#    * 45%  — трофейная мототехника;
	#    * 15%  — трофейные танки и артиллерия;
	#    * при наличии людских резервов и достатка вооружения в
	#      запасах — формируется новая дивизия «Трофейный отряд РПА»
	#      на текущей территории Гуляйполя (культурамм 30 дней).
	#  Условие «у захватившей дивизии есть ремонтная рота» в рамках
	#  документированных триггеров реализовано как: изучена технология
	#  ремонтной роты (tech_maintenance_company) И в армии страны есть
	#  шаблон с maintenance_company (has_template_containing_unit --
	#  точный пословный синтаксис из ванильного armor.txt). Попроверить
	#  состав КОНКРЕТНОЙ дивизии-победительницы движок не позволяет:
	#  триггеров на под-юниты дивизии не существует (см. triggers.info).
	#  Сам шаблон трофейного отряда ремонтную роту содержит.
	# ==============================================================
	on_army_leader_won_combat = {
		effect = {
			if = {
				limit = {
					FROM = {
						tag = GLP
						has_war = yes
					}
				}
				FROM = {
					# --- 50%: стрелковое вооружение и снаряжение ---
					if = {
						limit = { random = 50 }
						add_equipment_to_stockpile = {
							type = infantry_equipment
							amount = 120
							producer = GLP
						}
						add_equipment_to_stockpile = {
							type = support_equipment
							amount = 25
							producer = GLP
						}
					}
					# --- 25%: мототехника ---
					if = {
						limit = { random = 25 }
						add_equipment_to_stockpile = {
							type = motorized_equipment
							amount = 20
							producer = GLP
						}
					}
					# --- 8%: танки и артиллерия ---
					if = {
						limit = { random = 8 }
						add_equipment_to_stockpile = {
							type = artillery_equipment
							amount = 10
							producer = GLP
						}
						if = {
							limit = { has_dlc = "No Step Back" }
							add_equipment_to_stockpile = {
								type = light_tank_chassis
								amount = 2
								producer = GLP
							}
						}
						else = {
							add_equipment_to_stockpile = {
								type = light_tank_equipment_1
								amount = 2
								producer = GLP
							}
						}
					}
					# --- формирование новой дивизии из трофеев ---
					if = {
						limit = {
							has_manpower > 8000
							has_equipment = { infantry_equipment > 400 }
							has_equipment = { support_equipment > 60 }
							NOT = { has_country_flag = glp_trophy_division_cooldown }
						}
						random_owned_state = {
							limit = { is_controlled_by = GLP }
							create_unit = {
								division = "division_template = \"Трофейный отряд РПА\" start_manpower_factor = 1.0 start_experience_factor = 0.3"
								owner = GLP
							}
						}
						# формирование честно расходует резервы (ванильный паттерн
						# отрицательных amount у add_equipment_to_stockpile)
						add_manpower = -4000
						add_equipment_to_stockpile = {
							type = infantry_equipment
							amount = -400
						}
						add_equipment_to_stockpile = {
							type = support_equipment
							amount = -60
						}
						set_country_flag = { flag = glp_trophy_division_cooldown days = 45 }
						country_event = { id = glp_trophy.1 hours = 6 }
					}
				}
			}
		}
	}
}
