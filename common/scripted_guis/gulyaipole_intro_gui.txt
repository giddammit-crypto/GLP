# Гуляй-Поле — scripted GUI кинематографической заставки (HOI4 1.19.2)
#
# Одно окно из interface/gulyaipole_intro_custom.gui показывается, пока
# установлен glp_show_cinematic_intro. Голос запускается событием
# glp_cinematic_intro.1, а не из GUI.

scripted_gui = {
	gulyaipole_cinematic_intro = {
		window_name = "gulyaipole_cinematic_intro_window"
		context_type = player_context

		visible = {
			tag = GLP
			is_ai = no
			has_country_flag = glp_show_cinematic_intro
		}

		effects = {
			start_campaign_button_click = {
				clr_country_flag = glp_show_cinematic_intro
			}
		}

		triggers = {
			start_campaign_button_click_enabled = {
				always = yes
			}
			start_campaign_button_visible = {
				always = yes
			}
		}
	}
}
