# Событие: Беглецы из РККА и архивы Задова (Эхо московских чисток 1937–1938)
country_event = {
	id = glp_bi.4
	title = glp_bi.4.t
	desc = glp_bi.4.d
	picture = GFX_news_event_GLP_counter_intel
	is_triggered_only = yes

	trigger = {
		tag = GLP
		SOV = { exists = yes }
	}

	# 1. Принять беглецов: получить опытных командиров и конструкторов, но вызвать ярость Москвы
	option = {
		name = glp_bi.4.a
		add_manpower = 8000
		army_experience = 25
		add_research_slot = 0
		add_timed_idea = { idea = GLP_idea_cryptanalysis days = 365 }
		SOV = {
			add_opinion_modifier = { target = GLP modifier = GLP_soviet_anarchy_resentment }
			country_event = { id = glp_bi.5 days = 5 }
		}
	}

	# 2. Отказать: сохранить хрупкий мир с СССР ценой разочарования левых радикалов
	option = {
		name = glp_bi.4.b
		add_stability = -0.04
		add_popularity = { ideology = neutrality popularity = -0.05 }
		SOV = {
			add_opinion_modifier = { target = GLP modifier = GLP_kharkov_pact_signed }
		}
		add_political_power = 30
	}

	# 3. Использовать КРО Задова: перехватить архивы ГПУ и организовать агентурную сеть
	option = {
		name = glp_bi.4.c
		add_political_power = 50
		add_stability = 0.03
		add_timed_idea = { idea = GLP_idea_agent_networks days = 365 }
		add_command_power = 20
	}
}
