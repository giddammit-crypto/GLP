	is_triggered_only = yes

	option = {
		name = glp_diplo.1.a
		ai_chance = {
			factor = 70
			modifier = {
				factor = 0.2
				has_government = fascism
			}
			modifier = {
				factor = 1.4
				has_government = communism
			}
		}
		give_guarantee = GLP
		give_military_access = GLP

		if = {
			limit = { is_in_faction = yes }
			add_to_faction = GLP
		}
		else = {
			create_faction = GLP_kharkov_pact
			add_to_faction = GLP
		}
		GLP = {
			set_country_flag = GLP_pact_with_sov
			country_event = { id = glp_diplo.11 hours = 4 }
		}
	}
	option = {
		name = glp_diplo.1.b
		ai_chance = { factor = 30 }
		GLP = {
			add_stability = -0.04
			add_political_power = -25
			country_event = { id = glp_diplo.12 hours = 4 }
		}
	}
}
