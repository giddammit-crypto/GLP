# Партнёр решает, давать ли гарантию и доступ Гуляйполю.
country_event = {
	id = glp_diplo.1
	title = glp_diplo.1.t
	desc = glp_diplo.1.d
	picture = GFX_news_event_GLP_rebirth
	is_triggered_only = yes

	option = {
		name = glp_diplo.1.a
		ai_chance = {
			factor = 70
			modifier = {
				factor = 0.2
				has_government = fascism
			}
			modifier = {
				factor = 1.4
				has_government = communism
			}
		}
		give_guarantee = GLP
		give_military_access = GLP
		GLP = { give_military_access = SOV }
		if = {
			limit = { is_in_faction = yes }
			add_to_faction = GLP
		}
		else = {
			create_faction = GLP_kharkov_pact
			add_to_faction = GLP
		}
		GLP = {
			set_country_flag = GLP_pact_with_sov
			if = {
				limit = { has_idea = GLP_hostile_encirclement }
				swap_ideas = {
					remove_idea = GLP_hostile_encirclement
					add_idea = GLP_idea_recognized_border
				}
			}
			country_event = { id = glp_diplo.11 hours = 4 }
		}
		GER = { country_event = { id = glp_diplo.21 hours = 12 } }
		ROM = { country_event = { id = glp_diplo.22 hours = 12 } }
		ENG = { country_event = { id = glp_diplo.23 hours = 12 } }
	}
	option = {
		name = glp_diplo.1.b
		ai_chance = { factor = 30 }
		GLP = {
			add_stability = -0.04
			add_political_power = -25
			# Отвергнутый пакт = год изоляционного сопротивления:
			# крепче оборона и война, но дороже хлеб и медленнее ПП.
			add_timed_idea = { idea = GLP_idea_isolated_resistance days = 365 }
			country_event = { id = glp_diplo.12 hours = 4 }
		}
	}
}
