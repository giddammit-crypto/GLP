	option = {
		name = glp_kro.100.a
		# Разгром логистики врага строго в чужих приграничных стейтах:

			limit = {
				has_war_with = GLP
			}
			every_owned_state = {
				limit = {
					NOT = { is_core_of = GLP }
					NOT = { is_owned_by = GLP }

						state = 202 # Киев (СССР)
						state = 221 # Полтава / Чернигов (СССР)

						state = 91  # Станиславов / Тарнополь (Польша)
					}
				}
				damage_building = {
					type = railways
					damage = 4
				}
				damage_building = {
					type = supply_node
					damage = 2
				}
			}
			add_stability = -0.08
			add_war_support = -0.05
		}

		# Пополнение сил и снаряжения партизанами
		add_political_power = 50
		add_war_support = 0.10
		add_manpower = 15000
		add_equipment_to_stockpile = {
			type = infantry_equipment
			amount = 1500
			producer = GLP
		}

		# Формирование 3 ударных партизанских дивизий
		random_owned_controlled_state = {
			prioritize = { 226 200 227 }
			create_unit = {
				division = "name = \"1-й Приграничный повстанческий полк КРО\" division_template = \"Партизанский заслон народной самообороны\" start_experience_factor = 0.5 start_equipment_factor = 1.0"
				owner = GLP
			}
			create_unit = {
				division = "name = \"2-й Степной рейдовый курень КРО\" division_template = \"Ударный тачаночный курень им. Батьки Махно\" start_experience_factor = 0.6 start_equipment_factor = 1.0"
				owner = GLP
			}
			create_unit = {
				division = "name = \"3-й Подпольный батальон Черной Гвардии\" division_template = \"Партизанский заслон народной самообороны\" start_experience_factor = 0.5 start_equipment_factor = 1.0"
				owner = GLP
			}
		}

		set_country_flag = glp_kro_insurgency_fired
		news_event = { id = glp_kro.101 hours = 6 }
	}
