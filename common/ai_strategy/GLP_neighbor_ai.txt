# Постоянное давление соседей на Вольную территорию.
# add_ai_strategy на старте задаёт импульс; этот файл держит фон.

SOV_contain_GLP = {
	allowed = { original_tag = SOV }
	enable = {
		country_exists = GLP
		GLP = { exists = yes }
		NOT = { has_war_with = GLP }
		NOT = { is_in_faction_with = GLP }
	}
	abort = {
		OR = {
			NOT = { country_exists = GLP }
			is_in_faction_with = GLP
		}
	}
	ai_strategy = {
		type = antagonize
		id = GLP
		value = 120
	}
	ai_strategy = {
		type = contain
		id = GLP
		value = 70
	}
}
